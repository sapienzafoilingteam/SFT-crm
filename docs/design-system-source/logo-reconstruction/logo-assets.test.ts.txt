import { describe, it, expect } from 'vitest';
import { readFile } from 'node:fs/promises';
import sharp from 'sharp';

const variants = ['original', 'burgundy', 'ink', 'white'];
describe('Distribuzione dei marchi SFT', () => {
  it('mantiene una sola geometria autonoma e accessibile in tutte le varianti', async () => {
    const geometries: string[] = [];
    for (const variant of variants) {
      const svg = await readFile(`public/brand/logos/sft-${variant}.svg`, 'utf8');
      const document = new DOMParser().parseFromString(svg, 'image/svg+xml');
      expect(document.querySelector('parsererror')).toBeNull();
      expect(document.querySelector('image, text, use, script, foreignObject')).toBeNull();
      expect(document.querySelector('title')?.textContent).toContain('Sapienza Foiling Team');
      expect(document.querySelectorAll('path')).toHaveLength(6);
      for (const path of document.querySelectorAll('path')) expect(path.getAttribute('d')).toMatch(/Z$/);
      geometries.push([...document.querySelectorAll('path,circle')].map(e => e.outerHTML).join(''));
    }
    expect(new Set(geometries).size).toBe(1);
  });
  it('esporta PNG trasparenti senza tagli e con la stessa sagoma per ogni colore', async () => {
    const masks: Buffer[] = [];
    for (const variant of variants) {
      const { data, info } = await sharp(`public/brand/exports/sft-${variant}.png`).raw().toBuffer({ resolveWithObject: true });
      expect(info.height).toBe(2560);
      expect(info.channels).toBe(4);
      const alpha = Buffer.alloc(info.width * info.height);
      for (let i = 0; i < alpha.length; i++) alpha[i] = data[i * 4 + 3];
      masks.push(alpha);
      for (let x = 0; x < info.width; x++) {
        expect(alpha[x]).toBe(0);
        expect(alpha[(info.height - 1) * info.width + x]).toBe(0);
      }
      for (let y = 0; y < info.height; y++) {
        expect(alpha[y * info.width]).toBe(0);
        expect(alpha[y * info.width + info.width - 1]).toBe(0);
      }
    }
    expect(masks.every(mask => mask.equals(masks[0]))).toBe(true);
  });
});
